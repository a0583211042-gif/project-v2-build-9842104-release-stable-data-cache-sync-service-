async function handleLogin() { const code = document.getElementById('codeInput').value.trim(); if (!code) { alert("נא להקליד קוד"); return; } try { // בדיקה מול המסמך "קודי הגדרות" בדיוק כפי שיצרת בפיירבייס const docRef = db.collection("groupData").doc("קודי הגדרות"); const docSnap = await docRef.get(); if (docSnap.exists) { const codesMap = docSnap.data(); if (codesMap[code]) { const userName = codesMap[code]; localStorage.setItem('loggedUser', userName); window.location.href = 'board.html'; // מעבר ללוח הראשי } else { alert("קוד שגוי!"); } } else { alert("מסמך הקודים לא נמצא בענן."); } } catch (e) { alert("שגיאת התחברות מול הענן."); } }
